import boto3
import pandas as pd
from io import StringIO

BUCKET = "xideralaws-curso-lisset"
KEY = "spotify-2023.csv"

s3 = boto3.client("s3")

# Descargar nuevamente el archivo
response = s3.get_object(
    Bucket=BUCKET,
    Key=KEY
)

# Leer los bytes
contenido = response["Body"].read()

# Convertir manualmente de latin-1 a texto
texto = contenido.decode("latin-1")

# Crear el DataFrame
df = pd.read_csv(StringIO(texto))
